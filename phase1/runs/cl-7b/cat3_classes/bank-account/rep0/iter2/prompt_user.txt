    assert account.get_balance() == 100

    account.close()

def test_bank_account_concurrent_access():
    account = BankAccount()
    account.open()
    account.deposit(100)

    def deposit_thread():
        account.deposit(50)

    def withdraw_thread():
        account.withdraw(50)

    t1 = Thread(target=deposit_thread)
    t2 = Thread(target=withdraw_thread)
    t1.start()
    t2.start()
    t1.join()
    t2.join()

    assert account.get_balance() == 0
    account.close()
