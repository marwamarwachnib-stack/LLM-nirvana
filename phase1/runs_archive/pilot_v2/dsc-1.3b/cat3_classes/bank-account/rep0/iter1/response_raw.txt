Here is a simple test suite for the BankAccount class using pytest:

```python
import pytest
from BankAccount import BankAccount

def test_open_account():
    account = BankAccount()
    account.open()
    assert account.is_open == True

def test_deposit_into_open_account():
    account = BankAccount()
    account.open()
    account.deposit(100)
    assert account.get_balance() == 100

def test_withdraw_from_open_account():
    account = BankAccount()
    account.open()
    account.deposit(100)
    account.withdraw(50)
    assert account.get_balance() == 50

def test_cannot_deposit_negative_amount():
    account = BankAccount()
    account.open()
    with pytest.raises(ValueError):
        account.deposit(-100)

def test_cannot_withdraw_more_than_balance():
    account = BankAccount()
    account.open()
    account.deposit(100)
    with pytest.raises(ValueError):
        account.withdraw(150)

def test_cannot_withdraw_negative_amount():
    account = BankAccount()
    account.open()
    with pytest.raises(ValueError):
        account.withdraw(-100)

def test_close_account():
    account = BankAccount()
    account.open()
    account.deposit(100)
    account.close()
    assert account.is_open == False
    assert account.get_balance() == 100

def test_cannot_open_already_open_account():
    account = BankAccount()
    account.open()
    with pytest.raises(ValueError):
        account.open()
```

In this test suite, we are testing the different methods of the BankAccount class. We are checking if the account is open, if we can deposit and withdraw money, and if we can't deposit or withdraw negative amounts. We also test if we can close the account and if we can't open an already open account.
