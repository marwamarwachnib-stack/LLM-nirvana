def annotate(minefield): #1
    if not minefield: #2
        return [] #3
    verify_board(minefield) #4
    row_len = len(minefield[0]) #5
    col_len = len(minefield) #6
    board = [list(row) for row in minefield] #7
#8
    for index1 in range(col_len): #9
        for index2 in range(row_len): #10
            if board[index1][index2] != ' ': #11
                continue #12
#13
            low = max(index2 - 1, 0) #14
            high = min(index2 + 2, row_len + 2) #15
            counts = minefield[index1][low:high].count('*') #16
#17
            if index1 > 0: #18
                counts += minefield[index1 - 1][low:high].count('*') #19
            if index1 < col_len - 1: #20
                counts += minefield[index1 + 1][low:high].count('*') #21
            if counts == 0: #22
                continue #23
#24
            board[index1][index2] = str(counts) #25
    return [''.join(row) for row in board] #26

def verify_board(minefield): #29
    # Rows with different lengths #30
    row_len = len(minefield[0]) #31
    if not all(len(row) == row_len for row in minefield): #32
        raise ValueError('The board is invalid with current input.') #33
#34
    # Unknown character in board #35
    character_set = set() #36
    for row in minefield: #37
        character_set.update(row) #38
    if character_set - set(' *'): #39
        raise ValueError('The board is invalid with current input.') #40
